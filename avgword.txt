from mrjob.job import MRJob
import re

wordRe = re.compile(r"[a-zA-Z]+")

class AvgWord(MRJob):
  def mapper(self,_,line):
    for word in wordRe.findall(line):
      first_char = word[0].lower()
      yield first_char,len(word)

  def reducer(self,char,lengths):
    len=list(lenths)
    count =len(len)
    total = sum(len)
    avg = total/count
    yield char,round(avg,2)

AvgWord.run()
